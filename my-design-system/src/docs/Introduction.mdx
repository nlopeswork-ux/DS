export const CopyCommand = ({ command }) => {
  const [copied, setCopied] = useState(false);
  const copy = () => {
    navigator.clipboard
      ?.writeText(command)
      .then(() => {
        setCopied(true);
        setTimeout(() => setCopied(false), 1600);
      })
      .catch(() => {}); // clipboard blocked: leave the label as "Copy" rather than claim success
  };
  return (
    <div className="flex items-center justify-between gap-xl rounded-lg bg-primary-solid py-lg pr-lg pl-xl">
      <pre className="m-0 overflow-x-auto bg-transparent p-0 text-sm text-white">
        <code>{command}</code>
      </pre>
      <Button
        hierarchy="Secondary gray"
        size="sm"
        iconLeading
        iconLeadingSwap={<Icon name={copied ? 'check' : 'content_copy'} size="sm" />}
        onClick={copy}
      >
        {copied ? 'Copied' : 'Copy'}
      </Button>
    </div>
  );
};
