{/* Searchable grid — filters every group by substring match, click-to-copy the `<Icon name="..." />` snippet. */}
export const IconLibrary = () => {
  const [query, setQuery] = useState('');
  const [copied, setCopied] = useState(null);

  const handleCopy = (name) => {
    navigator.clipboard?.writeText(`<Icon name="${name}" />`);
    setCopied(name);
    setTimeout(() => setCopied((current) => (current === name ? null : current)), 1200);
  };

  const normalizedQuery = query.trim().toLowerCase();
  const filteredGroups = iconGroups
    .map((group) => ({
      ...group,
      icons: group.icons.filter((name) => name.includes(normalizedQuery)),
    }))
    .filter((group) => group.icons.length > 0);

  return (
    <div>
      <input
        type="text"
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        placeholder="Search icons by name…"
        className="mb-8 w-full rounded-md border border-primary bg-primary px-4 py-2.5 text-sm text-primary placeholder:text-quaternary focus-visible:outline-none focus-visible:ring-4 focus-visible:ring-brand-500"
      />

      {filteredGroups.length === 0 ? (
        <p className="text-sm text-tertiary">No icon in this sample matches "{query}" — any valid Material Symbols name still works with the component even if it isn't listed here.</p>
      ) : (
        filteredGroups.map((group) => (
          <div key={group.label} className="mb-10">
            <span className="mb-4 block text-sm font-semibold text-primary">{group.label}</span>
            <div className="grid grid-cols-3 gap-3 sm:grid-cols-4 md:grid-cols-6">
              {group.icons.map((name) => (
                <button
                  key={name}
                  type="button"
                  onClick={() => handleCopy(name)}
                  title={`Click to copy <Icon name="${name}" />`}
                  className="flex flex-col items-center gap-2 rounded-lg border border-secondary bg-secondary px-3 py-4 text-center transition-colors hover:border-brand hover:bg-brand-primary"
                >
                  <Icon name={name} size="lg" className="text-primary" />
                  <span className="font-mono text-tertiary" style={{ fontSize: '11px', wordBreak: 'break-all' }}>
                    {copied === name ? 'Copied!' : name}
                  </span>
                </button>
              ))}
            </div>
          </div>
        ))
      )}
    </div>
  );
};

# Icons

The Design System's single, official icon set — [Google Material Symbols](https://fonts.google.com/icons), a variable font loaded once globally (`.storybook/preview-head.html` / `index.html`). No SVG library, no emoji — every icon in the product goes through the [`Icon`](?path=/docs/components-icon--docs) component and a name from this catalog.
