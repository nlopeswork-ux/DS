{/*
  Primitive color ramp — responsive `auto-fill` grid (never a cramped
  horizontal strip): each step is its own cell with a square swatch, Step
  in bold and Hex in mono, with no repeated column labels (replaces the
  addon-docs native `<ColorPalette>`/`<ColorItem>`, which prints
  "Name"/"Swatches" for every group).
*/}
export const ColorRamp = ({ title, subtitle, colors }) => (
  <>
    <span className="mt-8 mb-3 block text-sm font-semibold text-primary">
      {title}{subtitle ? ` ${subtitle}` : ''}
    </span>
    <div className="color-ramp-grid">
      {Object.entries(colors).map(([step, hex]) => (
        <div key={step} className="flex flex-col items-start">
          <div
            className="w-full"
            style={{ height: '48px', borderRadius: '6px', border: '1px solid rgba(0,0,0,0.08)', backgroundColor: hex }}
          />
          <div className="mt-1.5 flex flex-col items-start w-full">
            <span className="font-bold text-primary" style={{ fontSize: '12px' }}>{step}</span>
            <span
              className="font-mono text-tertiary w-full"
              style={{ fontSize: '11px', marginTop: '2px', wordBreak: 'break-all', overflowWrap: 'anywhere' }}
            >
              {hex.startsWith('#') ? hex.toUpperCase() : hex.startsWith('rgba') ? `RGBA${hex.slice(4)}` : hex}
            </span>
          </div>
        </div>
      ))}
    </div>
  </>
);
